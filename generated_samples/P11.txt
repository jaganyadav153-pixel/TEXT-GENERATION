import string

def clean_text(text: str) -> list[str]:
    """Lowercase, remove punctuation, and split into words.

    Args:
        text: Raw input string.

    Returns:
        List of lowercase words with punctuation removed.
    """
    lowered = text.lower()
    no_punct = lowered.translate(str.maketrans("", "", string.punctuation))
    return no_punct.split()

# Example usage:
if __name__ == "__main__":
    print(clean_text("Hello, World! This is GPT-Test."))
    # ['hello', 'world', 'this', 'is', 'gpttest']
